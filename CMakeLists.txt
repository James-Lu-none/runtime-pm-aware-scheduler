cmake_minimum_required(VERSION 3.16)
project(runtime_pm_aware_scheduler LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PM_BUILD_RUNNER "Build ncnn workload runner" ON)
option(PM_VULKAN "Build vendored ncnn with Vulkan" ON)
option(PM_SYSTEM_NCNN "Use an installed ncnn package" OFF)
add_library(pm_policy INTERFACE)
target_include_directories(pm_policy INTERFACE include)
enable_testing()
add_executable(policy_test tests/policy_test.cpp)
target_link_libraries(policy_test PRIVATE pm_policy)
add_test(NAME policy_test COMMAND policy_test)
if(PM_BUILD_RUNNER)
  if(PM_SYSTEM_NCNN)
    find_package(ncnn CONFIG REQUIRED)
  else()
    if(NOT EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/ncnn/CMakeLists.txt")
      message(FATAL_ERROR "Run git submodule update --init --recursive")
    endif()
    set(NCNN_VULKAN ${PM_VULKAN} CACHE BOOL "" FORCE)
    set(NCNN_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
    set(NCNN_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
    set(NCNN_BUILD_BENCHMARK OFF CACHE BOOL "" FORCE)
    set(NCNN_BUILD_TESTS OFF CACHE BOOL "" FORCE)
    add_subdirectory(third_party/ncnn)
  endif()
  add_executable(pm_runner src/main.cpp src/backend.cpp)
  target_link_libraries(pm_runner PRIVATE pm_policy ncnn)
  target_compile_options(pm_runner PRIVATE -Wall -Wextra -Wpedantic)
endif()
